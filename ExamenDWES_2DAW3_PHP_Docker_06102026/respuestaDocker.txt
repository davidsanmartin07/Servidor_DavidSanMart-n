// Responde a la pregunta sobre docker del enunciado

docker run: nos permite ejecutar un comando en un contenedor.
-d o -deteach: Ejecutar el contenedor en segundo plano e imprimir el ID del contenedor.
--name: Asigna un nombre al contenedor.
-p o --publish: Publicar todos los puertos de el contenedor en un host.
